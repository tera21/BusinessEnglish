        const DEFAULT_CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vS-QrIEWFEMVfZmSGMxBAdAuSJgqNlF86iDhWZa2EcdvmSH9I3D75td5a4Ugxh1_oQd_FMdJs-NAm3t/pub?output=csv";


        let masterOriginalData = []; // 원본 전체 데이터 보관용

        let currentRoundPool = [];   // 현재 라운드에서 아직 못 맞춘 단어들의 원본 풀

        let words = [];              // 현재 라운드에서 가중치(난이도)가 반영되어 섞인 퀴즈 큐

        let currentIndex = 0;

        let currentRound = 1;

        let isFlipped = false;

        let isProcessing = false;


        window.onload = () => {

            let savedUrl = localStorage.getItem('sheetCsvUrl');

            if (!savedUrl) {

                savedUrl = DEFAULT_CSV_URL;

                localStorage.setItem('sheetCsvUrl', savedUrl);

            }

            fetchCSVData(savedUrl);

        };


        function parseCSV(text) {

            const lines = text.split('\n');

            const result = [];

            for (let i = 1; i < lines.length; i++) {

                const line = lines[i].trim();

                if (!line) continue;


                const row = [];

                let insideQuote = false;

                let currentItem = '';

                for (let c = 0; c < line.length; c++) {

                    let char = line[c];

                    if (char === '"') {

                        insideQuote = !insideQuote;

                    } else if (char === ',' && !insideQuote) {

                        row.push(currentItem.trim().replace(/^"|"$/g, ''));

                        currentItem = '';

                    } else {

                        currentItem += char;

                    }

                }

                row.push(currentItem.trim().replace(/^"|"$/g, ''));


                const w = row[0] || '';

                const m = row[1] || '';

                let diff = Number(row[2]) || 1; // 3번째 열 난이도 (기본값 1)

                if (diff < 1) diff = 1;


                if (w) {

                    result.push({ word: w, meaning: m, difficulty: diff });

                }

            }

            return result;

        }


        // 난이도(difficulty) 가중치 적용하여 출제 큐 생성

        function createWeightedQueue(dataPool) {

            let weightedList = [];

            dataPool.forEach(item => {

                let weight = item.difficulty;

                for (let i = 0; i < weight; i++) {

                    weightedList.push({ ...item });

                }

            });

            return shuffleArray(weightedList);

        }


        async function fetchCSVData(url) {

            setStatus('불러오는 중...', 'text-blue-500');

            try {

                const cacheBuster = `&t=${new Date().getTime()}`;

                const fetchUrl = url.includes('?') ? url + cacheBuster : url + '?' + cacheBuster.substring(1);


                const response = await fetch(fetchUrl);

                if (!response.ok) throw new Error('네트워크 응답 에러');

                

                const text = await response.text();

                const data = parseCSV(text);


                if (data && data.length > 0) {

                    masterOriginalData = data;

                    currentRound = 1;

                    currentRoundPool = [...masterOriginalData];

                    words = createWeightedQueue(currentRoundPool);

                    currentIndex = 0;

                    isFlipped = false;

                    document.querySelector('.card-container').classList.remove('flipped');

                    updateUI();

                    setStatus('정상 연동됨', 'text-emerald-600');

                } else {

                    throw new Error('데이터가 비어있습니다.');

                }

            } catch (error) {

                console.error('Fetch Error:', error);

                setStatus('연동 오류', 'text-red-500');

            }

        }


        function setStatus(text, colorClass) {

            const statusEl = document.getElementById('statusText');

            statusEl.innerText = text;

            statusEl.className = `text-xs font-semibold ${colorClass}`;

        }


        function updateUI() {

            if (words.length === 0) return;

            document.getElementById('roundBadge').innerText = `${currentRound}라운드`;

            document.getElementById('progressText').innerText = `${currentIndex + 1} / ${words.length} (총 ${currentRoundPool.length}단어)`;

            document.getElementById('meaningText').innerText = words[currentIndex].meaning;

            document.getElementById('wordText').innerText = words[currentIndex].word;

        }


        function toggleCard() {

            if (isProcessing) return;

            const container = document.querySelector('.card-container');

            isFlipped = !isFlipped;

            if (isFlipped) {

                container.classList.add('flipped');

            } else {

                container.classList.remove('flipped');

            }

        }


        function nextWord(isKnown) {

            if (isProcessing) return;


            const container = document.querySelector('.card-container');

            isProcessing = true;


            // 알고 맞춘 단어라면 현재 라운드 풀에서 제거 대상 기록

            let currentWordObj = words[currentIndex];

            if (isKnown) {

                // 현재 라운드 풀에서 해당 단어 제거 (중복 가중치 항목이 여러 개 있을 수 있으므로 동일 단어 모두 제거)

                currentRoundPool = currentRoundPool.filter(item => item.word !== currentWordObj.word);

            }


            // 카드가 앞면이면 먼저 뒤집어서 정답 보여주기

            isFlipped = false;

            container.classList.remove('flipped');


            setTimeout(() => {

                isFlipped = true;

                container.classList.add('flipped');

            }, 50);


            const waitTime = isKnown ? 700 : 1400;


            setTimeout(() => {

                proceedToNext();

                isProcessing = false;

            }, waitTime);

        }


        function proceedToNext() {

            currentIndex++;

            

            // 현재 라운드의 큐가 끝났을 때

            if (currentIndex >= words.length) {

                if (currentRoundPool.length === 0) {

                    alert('🎉 모든 단어를 완벽하게 맞췄습니다! 처음부터 다시 시작합니다.');

                    currentRound = 1;

                    currentRoundPool = [...masterOriginalData];

                } else {

                    currentRound++;

                    alert(`✨ ${currentRound - 1}라운드가 끝났습니다! 맞춘 단어는 제외하고 남은 ${currentRoundPool.length}개의 단어로 ${currentRound}라운드를 시작합니다.`);

                }

                

                // 새로운 라운드 큐 생성

                words = createWeightedQueue(currentRoundPool);

                currentIndex = 0;

            }

            

            isFlipped = false;

            document.querySelector('.card-container').classList.remove('flipped');

            

            setTimeout(() => {

                updateUI();

            }, 150);

        }


        function shuffleArray(array) {

            let curId = array.length;

            while (0 !== curId) {

                let randId = Math.floor(Math.random() * curId);

                curId -= 1;

                let tmp = array[curId];

                array[curId] = array[randId];

                array[randId] = tmp;

            }

            return array;

        }


        function openSettings() {

            const savedUrl = localStorage.getItem('sheetCsvUrl') || DEFAULT_CSV_URL;

            document.getElementById('sheetUrlInput').value = savedUrl;

            document.getElementById('settingsModal').classList.remove('hidden');

        }


        function closeSettings() {

            document.getElementById('settingsModal').classList.add('hidden');

        }


        function saveSettings() {

            const url = document.getElementById('sheetUrlInput').value.trim();

            if (url) {

                localStorage.setItem('sheetCsvUrl', url);

                closeSettings();

            } else {

                alert('URL을 입력해주세요.');

            }

        }


        function shareLink() {

            if (navigator.share) {

                navigator.share({

                    title: '모바일 영단어 퀴즈',

                    url: window.location.href,

                }).catch(console.error);

            } else {

                alert('현재 브라우저에서 공유 기능을 지원하지 않습니다.');

            }

        }
